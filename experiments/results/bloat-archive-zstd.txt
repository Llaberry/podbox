pub fn sweep_scaffold() -> String {
    let empty: &[u8] = &[];
    let dec = flate2::read::GzDecoder::new(empty);
    let mut ar = tar::Archive::new(dec);
    let n = ar.entries().map(|es| es.count()).unwrap_or(0);
    // The zstd arm T-0907 asks to price separately: zstd layers exist and are
    // not universal, so a named refusal is a legitimate v1 answer.

    format!("entries={n} zstd_ok={z}")
}
