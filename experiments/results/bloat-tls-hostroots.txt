pub fn sweep_scaffold() -> String {
    // ⚠ Host bundle ONLY: no webpki-roots. The file may not exist, which is the
    // cost this arm measures against bundling. T-0905.
    let mut roots = rustls::RootCertStore::empty();
    let pem = std::fs::read("/etc/ssl/certs/ca-certificates.crt").unwrap_or_default();
    let mut rd = std::io::BufReader::new(&pem[..]);
    for der in rustls_pemfile::certs(&mut rd).flatten() {
        let _ = roots.add(der);
    }
    let n = roots.len();
    let cfg = rustls::ClientConfig::builder()
        .with_root_certificates(roots)
        .with_no_client_auth();
    let name: rustls::pki_types::ServerName =
        "index.docker.io".to_string().try_into().expect("literal");
    let conn = rustls::ClientConnection::new(std::sync::Arc::new(cfg), name);
    format!("roots={n} handshake_startable={}", conn.is_ok())
}
